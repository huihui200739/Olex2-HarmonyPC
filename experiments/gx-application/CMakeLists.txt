cmake_minimum_required(VERSION 3.24)
project(olex2_gx_platform_components LANGUAGES CXX)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(gx_texture_image_cpu STATIC native/texture_image.cpp)
target_include_directories(gx_texture_image_cpu PUBLIC native)
add_executable(olex2_texture_image_cpu native/tests/texture_image_smoke.cpp)
target_link_libraries(olex2_texture_image_cpu PRIVATE gx_texture_image_cpu)
enable_testing()
add_test(NAME ordinary_texture_image_cpu COMMAND olex2_texture_image_cpu)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_library(gx_ohos_texture_image STATIC native/ohos_texture_image.cpp)
  target_link_libraries(gx_ohos_texture_image PUBLIC gx_texture_image_cpu PRIVATE image_source pixelmap)
endif()
add_executable(olex2_image_resource_audit native/tests/image_resource_audit_smoke.cpp)
target_link_libraries(olex2_image_resource_audit PRIVATE gx_texture_image_cpu)
add_test(NAME image_resource_audit COMMAND olex2_image_resource_audit "${CMAKE_BINARY_DIR}/resource-audit-artifacts")
add_executable(olex2_image_memory_scope native/tests/image_memory_scope_smoke.cpp)
target_link_libraries(olex2_image_memory_scope PRIVATE gx_texture_image_cpu)
add_test(NAME image_memory_scope COMMAND olex2_image_memory_scope "${CMAKE_BINARY_DIR}/memory-scope-artifacts")
