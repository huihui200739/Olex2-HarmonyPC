cmake_minimum_required(VERSION 3.24)
project(olex2_png_texture_candidate LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/Olex2PinnedPng.cmake")
add_olex2_pinned_png("${CMAKE_CURRENT_SOURCE_DIR}")
get_filename_component(GX_NATIVE "${CMAKE_CURRENT_SOURCE_DIR}/../gx-application/native" ABSOLUTE)
add_library(olex2_png_texture STATIC native/png_texture_decoder.c native/png_texture_image.cpp "${GX_NATIVE}/texture_image.cpp")
target_include_directories(olex2_png_texture PUBLIC native "${GX_NATIVE}")
target_link_libraries(olex2_png_texture PRIVATE png_static)
option(OLEX2_PNG_TESTS "Build ordinary synthetic PNG tests" ON)
if(OLEX2_PNG_TESTS)
  enable_testing()
  add_executable(olex2_png_texture_test tests/png_texture_smoke.cpp)
  target_link_libraries(olex2_png_texture_test PRIVATE olex2_png_texture png_static)
  add_test(NAME exact_png_texture_cpu COMMAND olex2_png_texture_test "${CMAKE_BINARY_DIR}/ordinary-png-fixtures")
endif()
