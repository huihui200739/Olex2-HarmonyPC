cmake_minimum_required(VERSION 3.24)
project(olex2_genuine_gx_catalog_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OLEX2_CANDIDATE "" CACHE PATH "Frozen candidate containing the genuine GX sources")
if(NOT EXISTS "${OLEX2_CANDIDATE}/native/platform_window_scene.h")
  message(FATAL_ERROR "Select the frozen persistent GX candidate")
endif()
include("${OLEX2_CANDIDATE}/cmake/Olex2Utility.cmake")
add_olex2_utility_library(gx_catalog_sdl "${OLEX2_CANDIDATE}/upstream")
include("${OLEX2_CANDIDATE}/cmake/Olex2Core.cmake")
add_olex2_core_library(gx_catalog_core gx_catalog_sdl "${OLEX2_CANDIDATE}/upstream")
file(GLOB GLIB_SOURCES CONFIGURE_DEPENDS "${OLEX2_CANDIDATE}/upstream/glib/*.cpp")
file(GLOB GX_SOURCES CONFIGURE_DEPENDS "${OLEX2_CANDIDATE}/upstream/gxlib/*.cpp")
list(REMOVE_ITEM GX_SOURCES "${OLEX2_CANDIDATE}/upstream/gxlib/wglscene.cpp" "${OLEX2_CANDIDATE}/upstream/gxlib/wxglscene.cpp")
add_library(gx_catalog_graphics STATIC ${GLIB_SOURCES} ${GX_SOURCES}
  "${OLEX2_CANDIDATE}/upstream/repository/IsoSurface.cpp"
  "${OLEX2_CANDIDATE}/native/original_interpreter.cpp"
  "${OLEX2_CANDIDATE}/native/original_log.cpp")
target_include_directories(gx_catalog_graphics PUBLIC "${OLEX2_CANDIDATE}/native" "${OLEX2_CANDIDATE}/upstream/glib" "${OLEX2_CANDIDATE}/upstream/gxlib")
find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
target_link_libraries(gx_catalog_graphics PUBLIC gx_catalog_core "${OPENGL_FRAMEWORK}")
add_executable(genuine_gx_catalog_host catalog_host.cpp)
target_link_libraries(genuine_gx_catalog_host PRIVATE gx_catalog_graphics)
enable_testing()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures")
add_test(NAME genuine_gx_catalog_host COMMAND genuine_gx_catalog_host "${CMAKE_CURRENT_BINARY_DIR}/fixtures/olex2-harmony")
