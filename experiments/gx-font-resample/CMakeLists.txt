cmake_minimum_required(VERSION 3.16)
project(original_resource_font_resampling LANGUAGES CXX)
add_library(resource_font_resample STATIC ../gx-font/native/resource_font_resample.cpp)
target_include_directories(resource_font_resample PUBLIC ../gx-font/native)
target_compile_features(resource_font_resample PUBLIC cxx_std_17)
target_compile_options(resource_font_resample PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
add_executable(resource_font_resample_check tests/resample.cpp)
target_link_libraries(resource_font_resample_check PRIVATE resource_font_resample)
target_compile_options(resource_font_resample_check PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_compile_options(resource_font_resample PRIVATE -Wno-unused-command-line-argument)
  target_compile_options(resource_font_resample_check PRIVATE -Wno-unused-command-line-argument)
else()
  enable_testing()
  add_test(NAME isolated_resource_font_resampling COMMAND resource_font_resample_check)
endif()

if(WX_IMAGE_REFERENCE_SOURCE AND NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_executable(resource_font_wx_reference_check tests/oracle.cpp "${WX_IMAGE_REFERENCE_SOURCE}")
  target_link_libraries(resource_font_wx_reference_check PRIVATE resource_font_resample)
  target_compile_options(resource_font_wx_reference_check PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
  add_test(NAME isolated_fixed_wx_source_rgb_reference COMMAND resource_font_wx_reference_check)
endif()

  target_compile_options(resource_font_wx_reference_cells PRIVATE -Wall -Wextra -Werror -pedantic -ffp-contract=off)
endif()
