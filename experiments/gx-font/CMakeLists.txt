cmake_minimum_required(VERSION 3.16)
project(harmony_native_glyph_raster LANGUAGES CXX)
add_library(original_font_cells STATIC native/original_font_cells.cpp)
target_include_directories(original_font_cells PUBLIC native)
target_compile_features(original_font_cells PUBLIC cxx_std_17)
target_compile_options(original_font_cells PRIVATE -Wall -Wextra -Werror -pedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_library(harmony_native_glyph_raster STATIC native/harmony_glyph_raster.cpp)
  target_include_directories(harmony_native_glyph_raster PUBLIC native)
  target_compile_features(harmony_native_glyph_raster PUBLIC cxx_std_17)
  # Only suppress the installed SDK's unused --gcc-toolchain driver argument.
  target_compile_options(original_font_cells PRIVATE -Wno-unused-command-line-argument)
  target_compile_options(harmony_native_glyph_raster PRIVATE -Wall -Wextra -Werror -pedantic -Wno-unused-command-line-argument)
  target_link_libraries(harmony_native_glyph_raster PUBLIC native_drawing original_font_cells)
  add_executable(harmony_glyph_link_check tests/link_check.cpp)
  target_link_libraries(harmony_glyph_link_check PRIVATE harmony_native_glyph_raster)
else()
  # This only verifies bounded pixel-cell packing, never fakes NativeDrawing.
  enable_testing()
  add_executable(original_font_cells_test tests/cells.cpp)
  target_link_libraries(original_font_cells_test PRIVATE original_font_cells)
  target_compile_options(original_font_cells_test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME isolated_original_font_cell_packing COMMAND original_font_cells_test)
  add_executable(original_font_route_test tests/routes.cpp)
  target_include_directories(original_font_route_test PRIVATE native)
  target_compile_features(original_font_route_test PRIVATE cxx_std_17)
  target_compile_options(original_font_route_test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME isolated_original_font_route_counters COMMAND original_font_route_test)
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_executable(original_font_description_test tests/descriptions.cpp)
  target_include_directories(original_font_description_test PRIVATE native)
  target_compile_features(original_font_description_test PRIVATE cxx_std_17)
  target_compile_options(original_font_description_test PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME isolated_original_vector_font_descriptions COMMAND original_font_description_test)
endif()
