cmake_minimum_required(VERSION 3.24)
project(olex2_texture_context LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OLEX2_TEXTURE_SOURCE "" CACHE PATH "Exact isolated 20-patch candidate upstream")
if(NOT EXISTS "${OLEX2_TEXTURE_SOURCE}/glib/glpixelstorescope.h")
  message(FATAL_ERROR "Select the isolated candidate; never edit production upstream")
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Olex2Utility.cmake")
add_olex2_utility_library(texture_context_sdl "${OLEX2_TEXTURE_SOURCE}")
add_executable(original_texture_context_mocked tests/original_texture_context_mocked.cpp
  "${OLEX2_TEXTURE_SOURCE}/glib/gltexture.cpp" "${OLEX2_TEXTURE_SOURCE}/glib/gloption.cpp")
target_include_directories(original_texture_context_mocked PRIVATE "${OLEX2_TEXTURE_SOURCE}/glib")
find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
target_link_libraries(original_texture_context_mocked PRIVATE texture_context_sdl "${OPENGL_FRAMEWORK}")
target_compile_definitions(original_texture_context_mocked PRIVATE OLEX2_HARMONY_GL4ES_TEXTURE_1D_READBACK=1)
enable_testing()
add_test(NAME original_texture_context_mocked COMMAND original_texture_context_mocked)
