cmake_minimum_required(VERSION 3.24)
project(olex2_texture_transfer_remediation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OLEX2_TEXTURE_SOURCE "" CACHE PATH "Exact isolated candidate upstream directory")
if(NOT EXISTS "${OLEX2_TEXTURE_SOURCE}/glib/glpixelstorescope.h")
  message(FATAL_ERROR "Select the remediated candidate, never run the unpatched upload route")
endif()
add_executable(pixel_store_scope tests/pixel_store_scope.cpp)
target_include_directories(pixel_store_scope PRIVATE "${OLEX2_TEXTURE_SOURCE}/glib")
enable_testing()
add_test(NAME pixel_store_scope COMMAND pixel_store_scope)
# The actual original manager is executed only with a deliberately mocked GL
# API in this host-only target. No replacement symbols enter the application.
if(APPLE AND NOT CMAKE_CROSSCOMPILING)
  include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/Olex2Utility.cmake")
  add_olex2_utility_library(texture_transfer_sdl "${OLEX2_TEXTURE_SOURCE}")
  add_executable(original_texture_manager_mocked tests/original_texture_manager_mocked.cpp
    "${OLEX2_TEXTURE_SOURCE}/glib/gltexture.cpp" "${OLEX2_TEXTURE_SOURCE}/glib/gloption.cpp")
  target_include_directories(original_texture_manager_mocked PRIVATE "${OLEX2_TEXTURE_SOURCE}/glib")
  find_library(OPENGL_FRAMEWORK OpenGL REQUIRED)
  target_link_libraries(original_texture_manager_mocked PRIVATE texture_transfer_sdl "${OPENGL_FRAMEWORK}")
  add_test(NAME original_texture_manager_mocked COMMAND original_texture_manager_mocked)
endif()
