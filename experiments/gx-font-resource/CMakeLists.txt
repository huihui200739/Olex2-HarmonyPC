cmake_minimum_required(VERSION 3.16)
project(original_olex_font_resource LANGUAGES C CXX)
if(NOT EXISTS "${OLEX2_RESOURCE_ZLIB_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Select the existing fixed candidate vendor/zlib using OLEX2_RESOURCE_ZLIB_ROOT")
endif()
set(ZLIB_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZLIB_BUILD_STATIC ON CACHE BOOL "" FORCE)
set(ZLIB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${OLEX2_RESOURCE_ZLIB_ROOT}" pinned-zlib)
add_library(olex_font_resource STATIC ../gx-font/native/olex_font_resource.cpp)
target_include_directories(olex_font_resource PUBLIC ../gx-font/native)
target_compile_features(olex_font_resource PUBLIC cxx_std_17)
target_compile_options(olex_font_resource PRIVATE -Wall -Wextra -Werror -pedantic)
target_link_libraries(olex_font_resource PUBLIC zlibstatic)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_compile_options(olex_font_resource PRIVATE -Wno-unused-command-line-argument)
endif()
add_executable(olex_font_resource_check tests/resources.cpp)
target_link_libraries(olex_font_resource_check PRIVATE olex_font_resource)
target_compile_options(olex_font_resource_check PRIVATE -Wall -Wextra -Werror -pedantic)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_compile_options(olex_font_resource_check PRIVATE -Wno-unused-command-line-argument)
else()
  enable_testing()
  add_test(NAME isolated_original_font_resource_codec COMMAND olex_font_resource_check)
endif()
