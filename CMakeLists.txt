cmake_minimum_required(VERSION 3.24)
project(olex2_harmonypc_baseline VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(OLEX2_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/upstream" CACHE PATH "Pinned Olex2 source checkout")

if(NOT EXISTS "${OLEX2_SOURCE_DIR}/sdl/defs.h")
  message(FATAL_ERROR "Olex2 source missing; see source.lock.json and README.md")
endif()

# OlexSys' own SDL utility layer, not Simple DirectMedia Layer.
include(cmake/Olex2Utility.cmake)
add_olex2_utility_library(olex2_sdl "${OLEX2_SOURCE_DIR}")
include(cmake/Olex2Core.cmake)
add_olex2_core_library(olex2_core olex2_sdl "${OLEX2_SOURCE_DIR}")
add_library(olex2_checks STATIC native/utility_checks.cpp)
target_include_directories(olex2_checks PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(olex2_checks PUBLIC olex2_sdl)
add_library(olex2_interpreter STATIC native/original_interpreter.cpp)
target_include_directories(olex2_interpreter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(olex2_interpreter PUBLIC olex2_sdl)
add_library(olex2_application_runtime STATIC native/application_runtime.cpp native/original_log.cpp)
target_link_libraries(olex2_application_runtime PUBLIC olex2_interpreter olex2_core)
add_executable(olex2_original_log_smoke native/tests/original_log_smoke.cpp)
target_link_libraries(olex2_original_log_smoke PRIVATE olex2_application_runtime)
add_executable(olex2_options_transfer_smoke native/tests/options_transfer_smoke.cpp)
target_link_libraries(olex2_options_transfer_smoke PRIVATE olex2_application_runtime)
add_executable(olex2_application_runtime_smoke native/tests/application_runtime_smoke.cpp)
target_link_libraries(olex2_application_runtime_smoke PRIVATE olex2_application_runtime olex2_desktop_commands)
add_library(olex2_desktop_commands STATIC native/desktop_commands.cpp)
target_include_directories(olex2_desktop_commands PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(olex2_desktop_commands PUBLIC olex2_interpreter)
add_executable(olex2_desktop_commands_smoke native/tests/desktop_commands_smoke.cpp)
target_link_libraries(olex2_desktop_commands_smoke PRIVATE olex2_desktop_commands)
add_library(olex2_mesh STATIC native/upstream_mesh.cpp native/primitive_batch.cpp native/primitive_triangles.cpp native/primitive_polygons.cpp native/primitive_lines.cpp native/primitive_line_paths.cpp)
target_include_directories(olex2_mesh PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(olex2_mesh PUBLIC olex2_sdl)
target_include_directories(olex2_mesh PUBLIC "${OLEX2_SOURCE_DIR}/glib")
add_library(olex2_color STATIC native/upstream_color.cpp "${OLEX2_SOURCE_DIR}/glib/gloption.cpp")
target_include_directories(olex2_color PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native" "${OLEX2_SOURCE_DIR}/glib")
target_link_libraries(olex2_color PUBLIC olex2_sdl)
# Original GL types are required here, but these targets never create a context.
add_executable(olex2_light_defaults_smoke native/tests/light_defaults_smoke.cpp "${OLEX2_SOURCE_DIR}/glib/gllight.cpp")
target_link_libraries(olex2_light_defaults_smoke PRIVATE olex2_color)
add_library(olex2_texture_header_compile OBJECT native/tests/texture_header_compile.cpp)
target_link_libraries(olex2_texture_header_compile PRIVATE olex2_color)
if(NOT APPLE)
  target_include_directories(olex2_light_defaults_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/glu/include" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/gl4es/include")
  target_include_directories(olex2_texture_header_compile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/glu/include" "${CMAKE_CURRENT_SOURCE_DIR}/vendor/gl4es/include")
endif()
add_library(olex2_material STATIC native/lit_row_reference.cpp native/material_commands.cpp native/upstream_material.cpp "${OLEX2_SOURCE_DIR}/glib/glmaterial.cpp")
target_include_directories(olex2_material PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native" "${OLEX2_SOURCE_DIR}/glib")
target_compile_definitions(olex2_material PUBLIC OLEX2_MATERIAL_VALUES_ONLY)
target_link_libraries(olex2_material PUBLIC olex2_color)
add_library(olex2_matrices STATIC native/original_matrices.cpp)
target_include_directories(olex2_matrices PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(olex2_matrices PUBLIC olex2_sdl)
add_executable(olex2_original_matrices_smoke native/tests/original_matrices_smoke.cpp)
target_link_libraries(olex2_original_matrices_smoke PRIVATE olex2_matrices)
add_executable(olex2_original_polygons_smoke native/tests/original_polygons_smoke.cpp)
target_link_libraries(olex2_original_polygons_smoke PRIVATE olex2_mesh)
add_executable(olex2_original_line_paths_smoke native/tests/original_line_paths_smoke.cpp)
target_link_libraries(olex2_original_line_paths_smoke PRIVATE olex2_mesh)
add_executable(olex2_original_lines_smoke native/tests/original_lines_smoke.cpp)
target_link_libraries(olex2_original_lines_smoke PRIVATE olex2_mesh)
add_executable(olex2_original_triangles_smoke native/tests/original_triangles_smoke.cpp)
target_link_libraries(olex2_original_triangles_smoke PRIVATE olex2_mesh)
add_executable(olex2_material_smoke native/tests/material_smoke.cpp)
target_link_libraries(olex2_material_smoke PRIVATE olex2_material)
add_executable(olex2_lit_material_smoke native/tests/lit_material_smoke.cpp)
target_link_libraries(olex2_lit_material_smoke PRIVATE olex2_material olex2_mesh)
add_executable(olex2_material_commands_smoke native/tests/material_commands_smoke.cpp)
target_link_libraries(olex2_material_commands_smoke PRIVATE olex2_material)
add_executable(olex2_color_smoke native/tests/color_smoke.cpp)
target_link_libraries(olex2_color_smoke PRIVATE olex2_color)
add_executable(olex2_mesh_smoke native/tests/mesh_smoke.cpp)
target_link_libraries(olex2_mesh_smoke PRIVATE olex2_mesh)
add_executable(olex2_submission_smoke native/tests/submission_smoke.cpp)
target_include_directories(olex2_submission_smoke PRIVATE "${OLEX2_SOURCE_DIR}/glib")
target_link_libraries(olex2_submission_smoke PRIVATE olex2_sdl)
# Desktop compatibility is a compile-only check, not a runtime GL claim.
if(APPLE AND NOT CMAKE_CROSSCOMPILING)
  add_library(olex2_desktop_glprimitive_compile OBJECT "${OLEX2_SOURCE_DIR}/glib/glprimitive.cpp")
  target_include_directories(olex2_desktop_glprimitive_compile PRIVATE "${OLEX2_SOURCE_DIR}/glib")
  target_link_libraries(olex2_desktop_glprimitive_compile PRIVATE olex2_sdl)
  add_library(olex2_desktop_material_commands_compile OBJECT native/tests/desktop_material_commands_compile.cpp)
  target_include_directories(olex2_desktop_material_commands_compile PRIVATE "${OLEX2_SOURCE_DIR}/glib")
  target_link_libraries(olex2_desktop_material_commands_compile PRIVATE olex2_sdl)
  add_library(olex2_desktop_material_compile OBJECT "${OLEX2_SOURCE_DIR}/glib/glmaterial.cpp")
  target_include_directories(olex2_desktop_material_compile PRIVATE "${OLEX2_SOURCE_DIR}/glib")
  target_link_libraries(olex2_desktop_material_compile PRIVATE olex2_sdl)
  add_library(olex2_desktop_glutil_compile OBJECT native/tests/desktop_glutil_compile.cpp)
  target_include_directories(olex2_desktop_glutil_compile PRIVATE "${OLEX2_SOURCE_DIR}/glib")
  target_link_libraries(olex2_desktop_glutil_compile PRIVATE olex2_sdl)
endif()
add_executable(olex2_primitive_smoke native/tests/primitive_smoke.cpp)
target_link_libraries(olex2_primitive_smoke PRIVATE olex2_mesh)
add_executable(olex2_sdl_smoke native/tests/sdl_smoke.cpp)
target_link_libraries(olex2_sdl_smoke PRIVATE olex2_checks)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  target_link_options(olex2_sdl_smoke PRIVATE -Wl,--no-undefined)
  add_executable(olex2_gl_link_probe EXCLUDE_FROM_ALL native/tests/gl_link_probe.cpp)
  target_link_libraries(olex2_gl_link_probe PRIVATE GLv4 EGL)
  target_link_options(olex2_gl_link_probe PRIVATE -Wl,--no-undefined)
  add_executable(olex2_gl_core_probe native/tests/gl_core_probe.cpp)
  target_link_libraries(olex2_gl_core_probe PRIVATE GLv4 EGL)
  target_link_options(olex2_gl_core_probe PRIVATE -Wl,--no-undefined)
  add_executable(olex2_gles_egl_smoke native/tests/gles_egl_smoke.cpp)
  target_link_libraries(olex2_gles_egl_smoke PRIVATE GLESv3 EGL)
  target_link_options(olex2_gles_egl_smoke PRIVATE -Wl,--no-undefined)
endif()
include(CTest)

  add_test(NAME light_defaults_smoke COMMAND olex2_light_defaults_smoke)
  add_test(NAME original_log_smoke COMMAND olex2_original_log_smoke "${CMAKE_CURRENT_BINARY_DIR}" "${OLEX2_SOURCE_DIR}/LICENCE.txt")
  add_test(NAME options_transfer_smoke COMMAND olex2_options_transfer_smoke "${CMAKE_CURRENT_BINARY_DIR}" "${OLEX2_SOURCE_DIR}/LICENCE.txt")
  add_test(NAME application_runtime_smoke COMMAND olex2_application_runtime_smoke "${CMAKE_CURRENT_BINARY_DIR}" "${OLEX2_SOURCE_DIR}/LICENCE.txt")
  add_test(NAME desktop_commands_smoke COMMAND olex2_desktop_commands_smoke)
  add_test(NAME original_polygons_smoke COMMAND olex2_original_polygons_smoke)
  add_test(NAME original_matrices_smoke COMMAND olex2_original_matrices_smoke)
