cmake_minimum_required(VERSION 3.20)
project(olex2_system_font_family LANGUAGES CXX)
if(CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_library(harmony_system_font_catalog STATIC native/system_font_catalog.cpp)
  target_include_directories(harmony_system_font_catalog PUBLIC native)
  target_compile_features(harmony_system_font_catalog PUBLIC cxx_std_17)
  target_compile_options(harmony_system_font_catalog PRIVATE -Wall -Wextra -Werror -pedantic -Wno-unused-command-line-argument)
  target_link_libraries(harmony_system_font_catalog PUBLIC native_drawing)
  add_executable(harmony_system_font_link_check tests/link_check.cpp)
  target_link_libraries(harmony_system_font_link_check PRIVATE harmony_system_font_catalog)
  set(OLEX2_BASE_FONT_HEADERS "${CMAKE_CURRENT_LIST_DIR}/../gx-font/native" CACHE PATH "Frozen original glyph sources")
  add_executable(harmony_system_font_device_probe tests/device_catalog_probe.cpp "${OLEX2_BASE_FONT_HEADERS}/harmony_glyph_raster.cpp" "${OLEX2_BASE_FONT_HEADERS}/original_font_cells.cpp")
  target_include_directories(harmony_system_font_device_probe PRIVATE "${OLEX2_BASE_FONT_HEADERS}")
  target_compile_options(harmony_system_font_device_probe PRIVATE -Wall -Wextra -Werror -pedantic -Wno-unused-command-line-argument)
  target_link_libraries(harmony_system_font_device_probe PRIVATE harmony_system_font_catalog)
else()
  enable_testing()
  add_executable(isolated_family_font_description tests/descriptions.cpp)
  set(OLEX2_BASE_FONT_HEADERS "${CMAKE_CURRENT_LIST_DIR}/../gx-font/native" CACHE PATH "Frozen original descriptor headers")
  target_include_directories(isolated_family_font_description PRIVATE native "${OLEX2_BASE_FONT_HEADERS}")
  target_compile_features(isolated_family_font_description PRIVATE cxx_std_17)
  target_compile_options(isolated_family_font_description PRIVATE -Wall -Wextra -Werror -pedantic)
  add_test(NAME isolated_family_font_description COMMAND isolated_family_font_description)
endif()
